# Adversarial snippet corpus for qa/diff-snippets.sh
#
# One hermetic snippet per line. No clock, no randomness, no paths
# outside a scratch dir, no network. Each line must be a complete
# program; stdout and exit status are compared against dash, with bash
# as a second opinion.
#
# Sections target the bug classes this QA round has actually produced:
# silently-wrong exit status, builtins that lie, and the engine dropping
# a command from a list.

# --- test/[: every unary predicate against a fixed truth table ---
test -e /etc/passwd
test -f /etc/passwd
test -d /etc
test -s /etc/passwd
test -r /etc/passwd
test -w /etc/passwd
test -x /etc/passwd
test -h /etc/passwd
test -L /etc/passwd
test -e /no/such/path_xyz
test -f /no/such/path_xyz
test -r /no/such/path_xyz
test -z ""
test -z "x"
test -n ""
test -n "x"
test ! -e /no/such/path_xyz
test -e /etc/passwd -a -d /etc
test -d /etc -a -f /etc/passwd
test -d /etc -o -e /no/such/path_xyz
test ! -d /etc

# --- test/[: binary predicates ---
test 1 -eq 1
test 1 -ne 1
test 1 -lt 2
test 2 -le 2
test 3 -gt 2
test 3 -ge 3
test abc = abc
test abc != abd
test a \< b
test b \> a
test -z "" -o -z ""

# --- test/[: arity corner cases (POSIX 2.6.1) ---
test
test ""
test x

test -t
test !
test -eq
test -a
test -o
test -nt
test -ef
test -z ""

test -eq 1
test -z -n

# --- test/[: parenthesised groups ---
test \( 1 -eq 1 \)
test \( 1 -eq 2 \)
test ! \( 1 -eq 2 \)
test \( 1 -eq 2 \) -o \( 3 -eq 3 \)

# --- && / || equal precedence, left-to-right (the dropped-command bug) ---
false && echo a || echo b
true || echo a && echo b
true && echo a || echo b
false || echo a && echo b
true && false || echo c
false && echo a || echo b || echo c
true || echo a || echo b
false || echo a || echo b
true && echo a && echo b
false && echo a && echo b
false && echo a || echo b && echo c
true || echo a && echo b && echo c
false && echo a && echo b || echo c
! false && echo a || echo b
false && echo a || true && echo b
false || false && echo a || echo b
true && true && false || echo z
false && echo a || false && echo b || echo c
echo pre && false || echo post

# --- && / || interacting with pipelines and redirects ---
false | true && echo x || echo y
true | false && echo x || echo y
echo a && echo b || echo c
{ echo grouped; } && echo after
(echo sub) && echo after-sub

# --- exit status plumbing ---
false; echo $?
true; echo $?
false && true; echo $?
true || false; echo $?
false; true; echo $?
true; false; echo $?
exit 3; echo unreachable
false; exit 4
(exit 5); echo $?
f() { return 7; }; f; echo $?
f() { false; }; f; echo $?
f() { return 0; }; f && echo ok || echo no

# --- subshell isolation (a classic POSIX tripwire) ---
(x=1); echo "[$x]"
x=1; (x=2); echo $x
(x=1); y=$x; echo "[$y]"
echo $( (echo sub) )
(cd /tmp); pwd > /dev/null; echo ok
(exit 0); echo $?

# --- command substitution status ---
x=$(false); echo $?
x=$(true); echo $?
echo "$(false)"; echo $?
[ "$(echo hi)" = hi ] && echo match
n=$(echo a; echo b); echo $n

# --- quoting and word splitting ---
x='  spaced  '; echo "[$x]"
IFS=:; x=a:b:c; set -- $x; echo $#
IFS=:; x=a:b:c; set -- $x; echo $1-$3
x="a b"; set -- $x; echo $#
x="a b"; set -- "$x"; echo $# "$1"
echo ${#PATH} > /dev/null; echo lenok
echo "a
b" | wc -l

# --- parameter expansion forms ---
x=; echo "${x:-def}"
x=; echo "${x-def}"
x=v; echo "${x:-def}"
x=v; echo "${x-def}"
x=v; echo "${x:+set}"
x=; echo "${x:+set}"
x=abcdef; echo "${x#a}" "${x##a*}"
x=abcdef; echo "${x%f}" "${x%%*f}"
unset y; echo "[${y:-fallback}]"
x=abc; echo "${x:1}"

# --- arithmetic ---

echo $((1<<4)) $((255>>4))
echo $((1|2)) $((3&1)) $((5^3))

echo $((2**3)) 2>/dev/null
n=0; n=$((n+1)); echo $n
echo $(( 07 + 1 ))

# --- redirections ---
echo x > /tmp/qa_r1; cat /tmp/qa_r1; rm -f /tmp/qa_r1

echo e 1>&2 2>/dev/null; echo done

EOF

no $expansion here

cat <<EOF > /tmp/qa_r4
redirected heredoc
EOF

exec 3>&1; echo via-fd3 >&3; exec 3>&-
(echo sub) > /tmp/qa_r5; cat /tmp/qa_r5; rm -f /tmp/qa_r5

# --- loops and control flow ---
i=0; while [ $i -lt 3 ]; do echo $i; i=$((i+1)); done
i=0; until [ $i -ge 3 ]; do echo $i; i=$((i+1)); done
for i in a b c; do echo $i; done
for i in; do echo never; done; echo after-empty-for
case abc in abc) echo m1;; *) echo m2;; esac
case xyz in abc) echo m1;; *) echo m2;; esac
case abc in a*) echo star;; esac
case "" in "") echo empty;; esac
if true; then echo t; else echo f; fi
if false; then echo t; else echo f; fi

# --- functions, locals, recursion guard ---
f() { echo "$1"; }; f one; f two
f() { echo "$#:$*"; }; f a b c
f() { echo "$@"; }; f a b
f() { return 0; }; if f; then echo ok; fi
g() { echo "$1" | tr a-z A-Z; }; g hi
h() { local_v=set; echo $local_v; }; h

# --- traps ---
trap 'echo trapped' EXIT; echo main
trap 'echo int' INT; echo main2
trap - INT; echo cleared
trap 'echo t' EXIT; trap 'echo u' EXIT; echo dup-trap

# --- misc builtins ---
cd /tmp && pwd
echo a b c | tr ' ' '-'
true | false; echo ${PIPESTATUS:-n/a} > /dev/null; echo pipeok

exec 3>&1; printf via-fd3 >&3; exec 3>&-; printf done

exec false; printf NOT-REACHED
exec 4>&2; exec 4>&-; printf closed

# --- case pattern groups: POSIX 2.6.4, parens are delimiters ---
case a in (a) printf m ;; esac
case ab in (ab) printf m ;; esac
case a in (*) printf any ;; esac
case a in (a|b) printf alt ;; esac
case a in (a) printf m ;; *) printf nomatch ;; esac
case zzz in (a) printf m ;; *) printf nomatch ;; esac
case a in (a*z) printf p ;; esac
case ab in (a*b) printf p ;; esac
case '[a]' in ([a]) printf br ;; esac

# --- for loops with the in-list omitted ---
set -- x y z; for i do printf %s $i; done

for i do printf empty ;; done; printf done
set -- only; for i do printf n=%s $i; done
